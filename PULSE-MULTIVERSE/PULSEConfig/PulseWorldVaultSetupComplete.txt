/*
  Backend replaces %%FIREBASE_TOKEN%% with the real Firebase custom token.
*/
const token = "%%FIREBASE_TOKEN%%";

function normalizeDest(d) {
  if (!d) return null;
  d = d.split("?")[0];
  d = d.replace(/^\//, "");
  d = d.replace(/\.html$/i, "");
  return d + ".txt";
}

document.addEventListener("DOMContentLoaded", async () => {
  const params = new URLSearchParams(window.location.search);
  const returnTo = params.get("returnTo");

  if (!token) {
    window.location.href = "/error.html";
    return;
  }

  // ⭐ v9 TOKEN + DEVICE
  localStorage.setItem("pw_token_v9", token);
  localStorage.setItem("pw_device_v9", "1");

  try {
    // Verify token + fetch identity
    const response = await fetch("https://verifytoken-ilx3agka5q-uc.a.run.app", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ token })
    });

    const data = await response.json();

    if (!data.success || !data.identity) {
      window.location.href = "/error.html";
      return;
    }

    // ⭐ v9 IDENTITY
    localStorage.setItem("pw_identity_v9", JSON.stringify(data.identity));

    // Preload avatar
    if (data.identity.photoURL) {
      const img = new Image();
      img.src = data.identity.photoURL;
    }

    // ⭐ Convert returnTo → OS route
    let destRoute = normalizeDest(returnTo);

    // ⭐ Save last route
    if (destRoute) {
      localStorage.setItem("pw_last_route", destRoute);
    }

    // ⭐ Redirect to OS
    setTimeout(() => {
      if (destRoute) {
        window.location.href = `?impulse=${encodeURIComponent(destRoute)}`;
      } else {
        window.location.href = "/";
      }
    }, 900);

  } catch (err) {
    window.location.href = "/error.html";
  }
});
