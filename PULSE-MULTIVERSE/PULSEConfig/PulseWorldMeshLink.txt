/* ---------------------------------------------------------
   BLUETOOTH ACTIVATION — FULL
--------------------------------------------------------- */
async function activateBluetoothMesh() {
  const log = document.getElementById("bluetooth-log");
  log.textContent = "PulseWorld: Requesting Bluetooth activation…";

  try {
    const device = await navigator.bluetooth.requestDevice({
      acceptAllDevices: true,
      optionalServices: ["battery_service"]
    });

    log.textContent += "\nPulseWorld: Bluetooth activated.";
    log.textContent += `\nPulseWorld: Device granted: ${device.name || "Unnamed device"}`;

    document.getElementById("mesh-bt-status").textContent = "active";

    return device;

  } catch (err) {
    log.textContent += `\nPulseWorld: Bluetooth activation failed: ${err}`;
    return null;
  }
}

/* ---------------------------------------------------------
   WAVE INTELLIGENCE
--------------------------------------------------------- */
function detectWaveIntelligence() {
  const conn = navigator.connection || navigator.webkitConnection || navigator.mozConnection;

  if (!conn) return;

  const eff = conn.effectiveType || "unknown";

  if (eff.includes("wifi")) {
    document.getElementById("mesh-wifi-status").textContent = "wifi";
  }
  if (eff.includes("4g") || eff.includes("5g")) {
    document.getElementById("mesh-cell-status").textContent = eff;
  }
  if (eff.includes("slow-2g")) {
    document.getElementById("mesh-cell-status").textContent = "2g";
  }

  document.getElementById("mesh-sat-status").textContent = "unknown";
  document.getElementById("mesh-tier-status").textContent = "tier-0";
}
