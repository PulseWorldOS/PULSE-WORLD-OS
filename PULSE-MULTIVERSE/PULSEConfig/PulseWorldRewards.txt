function appendBankMessage(area, role, text) {
  const bubble = document.createElement("div");
  bubble.style.marginBottom = "6px";
  bubble.style.fontSize = "13px";
  bubble.style.lineHeight = "1.4";
  bubble.style.color = role === "user" ? "#f9f9f9" : "#a7f5e4";
  bubble.textContent = (role === "user" ? "You: " : "Bank: ") + text;
  area.appendChild(bubble);
  area.scrollTop = area.scrollHeight;
}

    appendBankMessage(
      bankLogArea,

};

    const text = input.value.trim();

    bubble.style.fontSize = "12px";
    bubble.style.lineHeight = "1.4";

currencyMode = "USD";

currencyRates = {
  USD: 1,
  EUR: 0.92,
  GBP: 0.78,

function convert(amountUSD) {
  const rate = currencyRates[currencyMode];
  return (amountUSD * rate).toFixed(currencyMode === "JPY" ? 0 : 2);

function updateBankPanels(balance) {
  // Convert string → number
  const available = Number(balance.availableBalance) || 0;
  const pending   = Number(balance.pendingBalance) || 0;
  const total     = available + pending;
  const reserve = total * 0.05

  // Update main balance panels
  document.getElementById("balanceWithdraw").textContent = `$${available.toFixed(2)}`;
  document.getElementById("balancePending").textContent   = `$${pending.toFixed(2)}`;
  document.getElementById("balanceMain").textContent     = `$${total.toFixed(2)}`;
  document.getElementById("balanceStripeReserve").textContent      = `$${reserve.toFixed(2)}`;

  // Update core panels
  document.getElementById("coreBurstStatus").textContent = `Design 2`;
  document.getElementById("coreRewardsInfluence").textContent   = (PulseRealm.PulseIdentity?.PulsePoints ?? PulseRealm.PulsePoints ?? 0)+` PulsePoints`;
  document.getElementById("coreTotalDeposits").textContent     = `$${total.toFixed(2)}`;

  // Update flow panels
  document.getElementById("flowStripeToPulse").textContent = "0 transfers";
  document.getElementById("flowPulseToBank").textContent = "0 transfers";

  // Update automation panels
  document.getElementById("coreAutomation").textContent = "Idle";
  document.getElementById("coreFlowsToday").textContent = "0";

  // Update last sync timestamp
  document.getElementById("syncLast").textContent =
    new Date().toLocaleTimeString();

  console.log(
    "%c[PULSEWORLD::BANK] Panels Updated",
    "color:#00FFAA; font-weight:bold;"
  );
}
