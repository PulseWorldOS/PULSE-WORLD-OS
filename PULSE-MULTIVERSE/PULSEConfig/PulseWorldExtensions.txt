    function scrollToSection(id) {
      const el = document.getElementById(id);
      if (!el) return;
      el.scrollIntoView({ behavior: "smooth", block: "start" });
    }

    (function setupTether() {
      const box = document.getElementById("tetherBox");
      const input = document.getElementById("tetherCode");
      const btn = document.getElementById("tetherActivateBtn");
      const status = document.getElementById("tetherStatus");

      if (!box || !input || !btn || !status) return;

      btn.addEventListener("click", () => {
        const code = input.value.trim();
        if (!code) {
          status.innerHTML = 'Status: <span style="color:' + "var(--danger)" + '">TetherCode required</span>';
          return;
        }

        // Tag the tetherCode for a 60-second loop
        console.log("[PulseBrowserOS] TetherCode Activated:", code);

        status.innerHTML =
          'Status: <span style="color:' +
          "var(--success)" +
          '">Active (60s loop)</span>';

        // Example: send to extension if available
        try {
          if (typeof chrome !== "undefined" && chrome.runtime && chrome.runtime.sendMessage) {
            chrome.runtime.sendMessage({
              type: "PULSE_TETHER_ACTIVATE",
              tetherCode: code,
              ttlSeconds: 60
            });
          }
        } catch (_) {
          // ignore
        }

        // After 60 seconds, hide the tether box
        setTimeout(() => {
          if (box) {
            box.style.opacity = "0";
            box.style.transform = "translateY(4px)";
            setTimeout(() => {
              box.style.display = "none";
            }, 260);
          }
        }, 60000);
      });
    })();
  </script>
