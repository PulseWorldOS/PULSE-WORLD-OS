async function requestBestCamera() {
  const constraintsList = [
    { video: { facingMode: "environment", width: 3840, height: 2160 } },
    { video: { facingMode: "environment", width: 1920, height: 1080 } },
    { video: { facingMode: "environment", width: 1280, height: 720 } },
    { video: { facingMode: "environment" } }
  ];

  for (const c of constraintsList) {
    try {
      const s = await navigator.mediaDevices.getUserMedia(c);
      const track = s.getVideoTracks()[0];
      PulseRealm.GetUserMediaTrack = track;
      log("Camera started with:", track.getSettings());
      return s;
    } catch (err) {
      warn("Failed constraints:", c, err);
    }
  }

  throw new Error("No camera available");
}

async function startCamera() {
  video = document.getElementById("t-video");
  if (!video) return warn("No #t-video element found");
  
  // Create QR canvas if missing
  qrCanvas = document.getElementById("t-qr-canvas") || (() => {
    const c = document.createElement("canvas");
    c.id = "t-qr-canvas";
    c.style.display = "none";
    document.body.appendChild(c);
    return c;
  })();

  qrCtx = qrCanvas.getContext("2d");

  try {
    stream = await requestBestCamera();
  } catch (err) {
    warn("Camera request failed:", err);
    showProfileBubble("Camera not available. Try another device?");
    return;
  }

  const track = stream.getVideoTracks()[0];
  const settings = track.getSettings();

  // 🔥 MATCH CANVAS TO REAL CAMERA RESOLUTION
  qrCanvas.width = settings.width;
  qrCanvas.height = settings.height;

  video.srcObject = stream;

  video.onloadedmetadata = () => {
    video.play();
    video.style.opacity = "1";
  };

  video.onplaying = () => {
    const cutout = document.querySelector(".t-cutout");
    if (cutout) cutout.style.opacity = "1";
  };

  // 🔥 TAP TO AUTOFOCUS (REAL DEVICES)
  const cameraWindow = document.querySelector(".t-camera-window");
  if (cameraWindow) {
    cameraWindow.addEventListener("click", async (e) => {
      if (!stream) return;

      spawnTapRipple(e.clientX, e.clientY);

      const track = stream.getVideoTracks()[0];
      const caps = track.getCapabilities?.() || {};

      if (caps.focusMode?.includes("continuous")) {
        try {
          await track.applyConstraints({ advanced: [{ focusMode: "continuous" }] });
          return log("Autofocus refreshed");
        } catch (e) {
          warn("Autofocus failed", e);
        }
      }

      try {
        await track.applyConstraints({ advanced: [{}] });
        log("Fallback refocus applied");
      } catch (e) {
        warn("Fallback refocus failed", e);
      }
    });
  }

  scanLoop = requestAnimationFrame(tick);
}

function startInactivityWatcher() {
  stopInactivityWatcher();

  inactivityInterval = setInterval(() => {
    const now = Date.now();
    const diff = now - lastActivity;

    const bubbleScanningMessages = [
      "Hold Steady… I’m Looking 👀",
      "Let me get a closer look… ✦",
      "Almost There… the Magic is Forming… ✨",
      "I’m Reading the Vibes… 🌴",
      "Just a little closer… I can feel it… 🔮",
      "Scanning the Moment… Stay Right There…",
      "Something’s Shimmering… Give me a sec… ✨",
      "I’m following the Glow… Hold Still…",
      "The Gateway is Tuning in… 🌺",
      "Let me Focus… this one Feels Interesting…"
    ];

    if (diff > INACTIVITY_THRESHOLD) {
      const msg =
        bubbleScanningMessages[
          Math.floor(Math.random() * bubbleScanningMessages.length)
        ];
      showProfileBubble(msg);
      PulseRealm.TropicPulseBubbleController.active();
      lastActivity = now;

      const chance = Math.random();
      if (chance < 0.15) {
        setTimeout(() => {
          showProfileBubble("Scanning again… 🔄");
          PulseRealm.TropicPulseBubbleController.active();
          PulseRealm.TropicPulseScanner.reset();
        }, 1200);
      }
    }
  }, 1000);
}

function stopInactivityWatcher() {
  if (inactivityInterval) {
    clearInterval(inactivityInterval);
    inactivityInterval = null;
  }
}

function stopCamera() {
  if (stream) {
    stream.getTracks().forEach((t) => t.stop());
    stream = null;
  }

  if (video) {
    video.srcObject = null;
    video.style.opacity = "0";
  }

  if (scanLoop) {
    cancelAnimationFrame(scanLoop);
    scanLoop = null;
  }
}

function isProbablyQR(text) {
  const badPatterns = /[^a-zA-Z0-9:/?=._-]/g;
  const weird = (text.match(badPatterns) || []).length;
  return weird < text.length * 0.2;
}

function tick() {
  if (!video || video.readyState !== video.HAVE_ENOUGH_DATA) {
    scanLoop = requestAnimationFrame(tick);
    return;
  }

  if (!qrCanvas || !qrCtx) {
    scanLoop = requestAnimationFrame(tick);
    return;
  }

  qrCanvas.width = video.videoWidth;
  qrCanvas.height = video.videoHeight;

  qrCtx.drawImage(video, 0, 0, qrCanvas.width, qrCanvas.height);

  const imageData = qrCtx.getImageData(0, 0, qrCanvas.width, qrCanvas.height);
  const code = jsQR(imageData.data, qrCanvas.width, qrCanvas.height, {
    inversionAttempts: "dontInvert"
  });

  if (code) {
    const text = code.data;

    if (!text || text.length < 6 || !isProbablyQR(text)) {
      PulseRealm.TropicPulseScanner.handleScan("unreadable");
      scanLoop = requestAnimationFrame(tick);
      return;
    }

    if (text.toLowerCase().includes("tropicpulse")) {
      PulseRealm.TropicPulseScanner.handleScan("tropic");
    } else {
      PulseRealm.TropicPulseScanner.handleScan("nonTropic");
    }

    stopCamera();
    return;
  }

  scanLoop = requestAnimationFrame(tick);
}
