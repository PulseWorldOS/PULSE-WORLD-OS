    // Simple local clock for HUD
    function updateClock() {
      const el = document.getElementById("hudClock");
      if (!el) return;
      const now = new Date();
      const h = String(now.getHours()).padStart(2, "0");
      const m = String(now.getMinutes()).padStart(2, "0");
      const s = String(now.getSeconds()).padStart(2, "0");
      el.textContent = `LOCAL: ${h}:${m}:${s} MST`;
    }

(async () => {
    updateClock();
    const adminEmailBtn =  document.getElementById("admin-emails");
    const pexBuilderBtn =  document.getElementById("admin-pex");

    }
    pexBuilderBtn.onclick = async () => {
      loadInternalPage("PulseWorldReality");
      location.href="#PEXBuilder";
    }
    setInterval(updateClock, 1000);
