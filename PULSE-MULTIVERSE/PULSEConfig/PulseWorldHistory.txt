// ---------------------------------------------------------
// PulseHistory Renderer
// ---------------------------------------------------------

function renderPulseHistory() {
  const hist = PulseRealm.PulseHistory || { stack: [], index: -1 };
  const listEl = document.getElementById("pulseHistoryList");
  const detailEl = document.getElementById("historyDetailContent");

  const entries = hist.stack || [];

  const total = entries.length;
  document.getElementById("historyTotalEntries").textContent = total;

  if (total === 0) {
    document.getElementById("historyWorldAvgLoadTime").textContent = "~2.00s";
    document.getElementById("historyAvgLoadTime").textContent = "—";
    document.getElementById("historyFastestPage").textContent = "—";
    document.getElementById("historySlowestPage").textContent = "—";
    document.getElementById("historyAvgPulseSync").textContent = "—";
    document.getElementById("historyGpuParticipation").textContent = "—";
    document.getElementById("historySavedTime").textContent = "—";
    return;
  }

  let totalTiming = 0;
  let fastest = entries[0];
  let slowest = entries[0];

  entries.forEach(e => {
    let timingEnvelope = e.timing ?? 300;
    totalTiming += timingEnvelope;
    if (timingEnvelope < fastest.timing && timingEnvelope >= 10 && fastest.timing >= 10 && timingEnvelope != 300) fastest = e;
    if (timingEnvelope > slowest.timing) slowest = e;
  });
  const avgTiming = (totalTiming / total);
  const savedTiming = 2 - (avgTiming/1000);
  const savedpercent = (savedTiming / 2)*100;

  document.getElementById("historyWorldAvgLoadTime").textContent = "~2.00s";
  document.getElementById("historyAvgLoadTime").textContent = `${avgTiming.toFixed(2)} ms`;
  document.getElementById("historyFastestPage").textContent = fastest.page;
  document.getElementById("historySlowestPage").textContent = slowest.page;
  document.getElementById("historyAvgPulseSync").textContent = `100%`;
  document.getElementById("historyGpuParticipation").textContent = `~5%`;
  document.getElementById("historySavedTime").textContent = `${savedpercent.toFixed(2)}%`;

  listEl.innerHTML = "";

  entries.forEach((entry, index) => {
    const div = document.createElement("div");
    div.className = "history-entry";
    div.dataset.index = index;
    
    const syncValue = entry.disableSW ? "45%" : "100%";
    div.innerHTML = `
      <div class="history-entry-left">
        <span class="history-page-name">${entry.page}</span>
        <span class="history-page-date">${entry.time}</span>
      </div>

      <div class="history-entry-right">
        <span class="history-load-time">Load Duration: ${(entry.timing).toFixed(2)} ms</span>
        <span class="history-pulse-sync">Pulse Sync: ${syncValue}</span>
        <button class="btn pulse-btnA history-jump-btn" data-index="${index}">Details</button>
      </div>
    `;
    
    div.addEventListener("click", () => showHistoryDetail(index));
    listEl.appendChild(div);
  });

  function showHistoryDetail(i) {
    const e = entries[i];
    
    const syncValue = e.disableSW ? "45%" : "100%";
    detailEl.innerHTML = `
      <div class="history-detail-row"><span>Page:</span> ${e.page}</div>
      <div class="history-detail-row"><span>Date:</span> ${e.time}</div>
      <div class="history-detail-row"><span>Index:</span> ${i}</div>
      <div class="history-detail-row"><span>Load Duration:</span> ${(e.timing).toFixed(2)} ms</div>
      <div class="history-detail-row"><span>Approx. View Duration:</span> ${(e.timeSpent).toFixed(2)} ms</div>
      <div class="history-detail-row"><span>Pulse Sync:</span> ${syncValue}</div>
      <div class="history-detail-row"><span>Environment:</span> Pulse World</div>
      <div class="history-detail-row"><span>World Layer:</span> PulseBrowser</div>
      <div class="history-detail-row"><span>Session ID:</span> ${e.signature}</div>
    `;
  }

  document.getElementById("historyExportBtn").onclick = () => {
    const blob = new Blob([JSON.stringify(hist, null, 2)], { type: "application/json" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = "PulseHistory.json";
    a.click();
    URL.revokeObjectURL(url);
  };

  document.getElementById("historyAnalyzeBtn").onclick = () => {
    alert("PulseHistory Analysis coming soon — GPU timing, sync drift, world continuity, and load physics.");
  };

  document.getElementById("historyClearBtn").onclick = () => {
    if (!confirm("Clear local PulseHistory? This cannot be undone.")) return;
    localStorage.removeItem("pulseHistory");
    PulseRealm.PulseHistory = { stack: [], index: -1 };
    renderPulseHistory();
  };
}
