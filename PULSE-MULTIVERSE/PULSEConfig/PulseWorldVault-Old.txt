async function loadEvents() {
  mark("VAULT: loadEvents");

  const htmlVersion = PulseRealm.globalCacheVersion.eventVersion || 1;
  const localVersion = Number(localStorage.getItem("eventCacheVersion") || 1);

  // HTML version newer → force reload
  if (htmlVersion > localVersion) {
    localStorage.setItem("forceReload_events", "1");
    localStorage.setItem("eventCacheVersion", htmlVersion);
  }

  const forceReload = localStorage.getItem("forceReload_events") === "1";

  // Load cached events
  const cachedRaw = localStorage.getItem("cachedUnifiedEvents");
  const cached = cachedRaw ? JSON.parse(cachedRaw) : null;

  const validCache =
    cached &&
    typeof cached === "object" &&
    Array.isArray(cached.all) &&
    Array.isArray(cached.today) &&
    Array.isArray(cached.upcoming);

  // USE CACHE IF VALID + NOT FORCE RELOAD
  if (!forceReload && validCache) {
    mark("VAULT: Using cached unified events");
    PulseRealm.allEvents      = cached.all;
    PulseRealm.eventsToday    = cached.today;
    PulseRealm.eventsUpcoming = cached.upcoming;
    PulseRealm.eventsNow      = cached.now;
    PulseRealm.eventsThisWeek = cached.thisWeek;
    PulseRealm.eventsWeekend  = cached.weekend;
    PulseRealm.eventsTonight  = cached.tonight;
    return cached;
  }

  // Otherwise → fetch fresh
  mark("VAULT: Fetching unified events from Cloud Function…");

  try {
    const data = await getHook({
      type: "getEvents"
    });

    const normalized = {
      all:      Array.isArray(data.all)      ? data.all      : [],
      today:    Array.isArray(data.today)    ? data.today    : [],
      upcoming: Array.isArray(data.upcoming) ? data.upcoming : [],
      now:      Array.isArray(data.now)      ? data.now      : [],
      thisWeek: Array.isArray(data.thisWeek) ? data.thisWeek : [],
      weekend:  Array.isArray(data.weekend)  ? data.weekend  : [],
      tonight:  Array.isArray(data.tonight)  ? data.tonight  : []
    };

    // Save to cache
    localStorage.setItem("cachedUnifiedEvents", JSON.stringify(normalized));
    localStorage.removeItem("forceReload_events");

    // Populate globals
    PulseRealm.allEvents      = normalized.all;
    PulseRealm.eventsToday    = normalized.today;
    PulseRealm.eventsUpcoming = normalized.upcoming;
    PulseRealm.eventsNow      = normalized.now;
    PulseRealm.eventsThisWeek = normalized.thisWeek;
    PulseRealm.eventsWeekend  = normalized.weekend;
    PulseRealm.eventsTonight  = normalized.tonight;

    return normalized;

  } catch (err) {
    error("❌ Failed to load unified events:", err);
    return null;
  }
}

async function loadLoyalty(uid) {
  try {
    // ---------------------------------------------
    // 1. LOAD PULSE IDENTITY FROM LOCAL STORAGE
    // ---------------------------------------------
    let rawIdentity = localStorage.getItem("PulseIdentity");
    let identity = rawIdentity ? JSON.parse(rawIdentity) : null;

    // If identity missing or wrong user → reset
    if (!identity || identity.bank !== uid) {
      identity = {
        bank,
        loyalty: {},
        wallet: {},
        photos: {}
      };
    }

    // ---------------------------------------------
    // 2. CHECK FOR LEGACY STANDALONE LOYALTY
    // ---------------------------------------------
    let legacyRaw = localStorage.getItem("PulseLoyalty");
    let legacy = legacyRaw ? JSON.parse(legacyRaw) : null;

    // ---------------------------------------------
    // 3. MERGE PRIORITY:
    // identity.loyalty > legacy loyalty > empty loyalty
    // ---------------------------------------------
    let loyalty = identity.loyalty && typeof identity.loyalty === "object"
      ? identity.loyalty
      : legacy && typeof legacy === "object"
        ? legacy
        : {};

    // ---------------------------------------------
    // 4. NORMALIZE LOYALTY FIELDS
    // ---------------------------------------------
    loyalty = {
      uid,
      pointsBalance: Number(loyalty.pointsBalance) || 0,
      lifetimePoints: Number(loyalty.lifetimePoints) || 0,
      history: loyalty.history || [],

      streakCount: Number(loyalty.streakCount) || 0,
      tier: loyalty.tier || "Seashell",

      seasonalActive: Boolean(loyalty.seasonalActive),
      seasonalMultiplier: Number(loyalty.seasonalMultiplier) || 1,
      seasonalName: loyalty.seasonalName || "",

      referralCode: loyalty.referralCode || "",
      referrals: loyalty.referrals || [],

      lastVaultVisit: loyalty.lastVaultVisit || null,
      lastEarnedDate: loyalty.lastEarnedDate || null,
      lastOrderDate: loyalty.lastOrderDate || null,
      totalOrders: Number(loyalty.totalOrders) || 0,
      vaultVisitCount: Number(loyalty.vaultVisitCount) || 0,
      lastActive: loyalty.lastActive || null
    };

    // ---------------------------------------------
    // 5. WRITE LOYALTY BACK INTO IDENTITY
    // ---------------------------------------------
    identity.loyalty = loyalty;

    // ---------------------------------------------
    // 6. SAVE UPDATED IDENTITY TO LOCAL STORAGE
    // ---------------------------------------------
    localStorage.setItem("PulseIdentity", JSON.stringify(identity));

    // ---------------------------------------------
    // 7. REMOVE LEGACY KEYS (CLEANUP)
    // ---------------------------------------------
    localStorage.removeItem("PulseLoyalty");
    localStorage.removeItem("cachedLoyalty");
    localStorage.removeItem("loyaltyCacheVersion");
    localStorage.removeItem("forceReload_loyalty");

    // ---------------------------------------------
    // 8. RETURN FINAL LOYALTY OBJECT
    // ---------------------------------------------
    return loyalty;

  } catch (err) {
    console.error("❌ loadLoyalty failed:", err);

    // Safe fallback loyalty
    const fallback = {
      uid,
      pointsBalance: 0,
      lifetimePoints: 0,
      history: [],
      tier: "Seashell",
      streakCount: 0,
      seasonalActive: false,
      seasonalMultiplier: 1,
      seasonalName: "",
      referralCode: "",
      referrals: []
    };

    return fallback;
  }
}

bubble.appendChild(opt);

function normalizeDateMs(value) {
if (!value) return null;

// Firestore Timestamp object
if (typeof value.toMillis === "function") {
return value.toMillis();
}

// Firestore server timestamp placeholder (not resolved yet)
if (value instanceof Date) {
const ms = value.getTime();
return isNaN(ms) ? null : ms;
}

// Firestore { _seconds, _nanoseconds }
if (typeof value === "object" && "_seconds" in value) {
return value._seconds * 1000 + Math.floor((value._nanoseconds || 0) / 1e6);
}

// Human-readable Firestore console string
// Example: "March 10, 2026 at 2:09:53 PM UTC-7"
if (typeof value === "string") {
const cleaned = value
  .replace(" at ", " ")
  .replace("UTC", "GMT"); // Date.parse understands GMT offsets

const ms = Date.parse(cleaned);
if (!isNaN(ms)) return ms;

// Fallback: try raw parse
const fallback = Date.parse(value);
return isNaN(fallback) ? null : fallback;
}

// Milliseconds number
if (typeof value === "number") {
return value > 0 ? value : null;
}

return null;
}

function isWalkable(userLoc, coords) {
if (!userLoc || !coords) return false;
const dist = getDistanceMeters(userLoc, coords);
return dist !== null && dist <= 1200;
}

function getDistanceMeters(userLoc, businessLoc) {
if (!userLoc || !businessLoc) return null;

const R = 6371000;
const toRad = d => (d * Math.PI) / 180;

const dLat = toRad(businessLoc.lat - userLoc.lat);
const dLon = toRad(businessLoc.lng - userLoc.lng);

const lat1 = toRad(userLoc.lat);
const lat2 = toRad(businessLoc.lat);

const a =
Math.sin(dLat / 2) ** 2 +
Math.cos(lat1) * Math.cos(lat2) * Math.sin(dLon / 2) ** 2;

return R * (2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a)));
}

async function handleDirectionsIntent(message, userLocation) {
if (!isDirectionsIntent(message)) return null;

const result = resolvePlace(
message,
PulseRealm.allBusinesses,
PulseRealm.allEvents
);

// SINGLE MATCH
if (result.mode === "single") {
const item = result.match.item;
const coords = item.coords || null;

return buildDirections(item, {
  type: result.match.type,
  walkable: isWalkable(userLocation, coords),
  distanceMeters: getDistanceMeters(userLocation, coords)
});
}

// ASK USER TO CHOOSE
if (result.mode === "ask") {
PulseRealm.__pendingPlaceChoices = result.matches;
return respondToResolution(result);
}

// NEED MORE DETAIL
if (result.mode === "refine") {
return {
  text: "✨ I found several possible matches. Try giving me a little more detail.",
  optionsHtml: ""
};
}

// GOOGLE FALLBACK
if (result.mode === "none") {
const google = await fetch(`/resolveVenue?q=${encodeURIComponent(message)}`)
  .then(r => r.json());

if (google.status === "ok") {
  return buildDirections(google, {
    type: "google",
    walkable: isWalkable(userLocation, google),
    distanceMeters: getDistanceMeters(userLocation, google)
  });
}

return {
  text: "✨ I couldn’t find anything that matches that.",
  optionsHtml: ""
};
}

return null;
}

function buildDirections(place, meta = {}) {
const { walkable, distanceMeters, type } = meta;

const dist = distanceMeters
? `${Math.round(distanceMeters)}m`
: "nearby";

const vibe = walkable
? "🚶‍♂️ A short walk away"
: "🛺 A quick golf cart ride";

// BUSINESS
if (type === "business" || place.busname) {
const name = place.busname || place.name;
const loc = place.location || place.address || "San Pedro";

return {
  text:
    `🗺️ <b>Directions to ${name}</b><br>` +
    `${vibe} — about <b>${dist}</b>.<br><br>` +
    `📍 ${loc}<br>` +
    `✨ I can guide you turn‑by‑turn soon.`,
  optionsHtml: ""
};
}

// EVENT
if (type === "event" || place.title || place.venue) {
const name =
  place.title ||
  place.venue ||
  (place.description && place.description.split(" ").slice(0, 3).join(" "));

const loc = place.address
  ? `${place.venue || place.title}, ${place.address}`
  : (place.venue || place.title);

return {
  text:
    `🗺️ <b>Directions to ${name}</b><br>` +
    `${vibe} — about <b>${dist}</b>.<br><br>` +
    `📍 ${loc}<br>` +
    `✨ I can guide you turn‑by‑turn soon.`,
  optionsHtml: ""
};
}

// GOOGLE MAPS
if (type === "google" || place.placeId) {

function matchField(query, text) {
  return sharesMeaningfulToken(query, text) && fuzzyMatchSafe(text, query);
}

function fuzzyMatchSafe(a, b) {
  if (!a || !b) return false;

  a = a.toLowerCase().replace(/[^a-z0-9]/g, "");
  b = b.toLowerCase().replace(/[^a-z0-9]/g, "");

  if (a.includes(b) || b.includes(a)) return true;

  if (Math.abs(a.length - b.length) > 3) return false;

  const dp = Array(b.length + 1).fill(null).map(() =>
    Array(a.length + 1).fill(null)
  );

  for (let i = 0; i <= a.length; i++) dp[0][i] = i;
  for (let j = 0; j <= b.length; j++) dp[j][0] = j;

  for (let j = 1; j <= b.length; j++) {
    for (let i = 1; i <= a.length; i++) {
      const cost = a[i - 1] === b[j - 1] ? 0 : 1;
      dp[j][i] = Math.min(
        dp[j][i - 1] + 1,
        dp[j - 1][i] + 1,
        dp[j - 1][i - 1] + cost
      );
    }
  }

  return dp[b.length][a.length] <= 2;
}

function resolvePlace(query, businesses, events) {
  const tiers = [
    // Tier 1 — Strongest
    { label: "Business Name", type: "business", field: "busname", list: businesses },
    { label: "Event Title", type: "event", field: "title", list: events },

    // Tier 2 — Medium
    { label: "Business Location", type: "business", field: "location", list: businesses },
    { label: "Event Venue", type: "event", field: "venue", list: events },

    // Tier 3 — Weak
    { label: "Business Description", type: "business", field: "description", list: businesses },
    { label: "Event Description", type: "event", field: "description", list: events }
  ];

  let candidates = [];

  for (const tier of tiers) {
    for (const item of tier.list) {
      const text = item[tier.field];
      if (matchField(query, text)) {
        candidates.push({ item, type: tier.type });
        if (candidates.length >= 3) break;
      }
    }

    if (candidates.length > 0) break; // stop at first tier with matches
  }

  // No matches
  if (candidates.length === 0) {
    return { mode: "none" };
  }

  // One match — return immediately
  if (candidates.length === 1) {
    return { mode: "single", match: candidates[0] };
  }

  // 2–3 matches — ask user
  if (candidates.length <= 3) {
    return { mode: "ask", matches: candidates };
  }

  // More than 3 — ask user to refine
  return { mode: "refine" };
}

function sharesMeaningfulToken(query, text) {
  if (!text) return false;

  const qTokens = tokenize(query);
  const tTokens = tokenize(text);

  return qTokens.some(qt =>
    qt.length >= 3 && tTokens.some(tt => tt.includes(qt) || qt.includes(tt))
  );
}

function tokenize(str) {
  return str
    .toLowerCase()
    .replace(/[^a-z0-9\s]/g, "")
    .split(/\s+/)
    .filter(Boolean);
}

async function checkReminders(userContext) {
  if (PulseRealm.__remindersChecked) return;
  PulseRealm.__remindersChecked = true;

  const uid = userContext.bank;
  if (!uid) return;

  try {
    // -----------------------------------------
    // ⭐ LOAD REMINDERS FROM FIRESTORE
    // -----------------------------------------
    const snap = await getDocs(
      collection(db, "Users", uid, "Reminders")
    );

    const reminders = snap.docs.map(doc => ({
      id: doc.id,
      ...doc.data()
    }));

    if (!reminders.length) return;

    const now = PulseRealm.PulseNOW;
    const triggered = [];

    // -----------------------------------------
    // ⭐ FILTER TRIGGERED REMINDERS
    // -----------------------------------------
    for (const r of reminders) {
      if (r.triggered) continue;

      if (r.type === "nextVaultOpen") {
        triggered.push(r);
      }

      if (r.type === "time" && r.triggerAt && r.triggerAt.toMillis() <= now) {
        triggered.push(r);
      }

      if (r.type === "nextAppOpen") {
        triggered.push(r);
      }
    }

    if (!triggered.length) return;

    // -----------------------------------------
    // ⭐ CLEAR PENDING STATE
    // -----------------------------------------
    const ctx = loadVaultContext() || {};
    ctx.pendingQuestion = null;
    ctx.pendingData = null;
    saveVaultContext(ctx);

    // -----------------------------------------
    // ⭐ DISPLAY REMINDERS
    // -----------------------------------------
    for (const r of triggered) {
      addBubble("Vault Spirit", r.message, false, {
        extraClasses: "reminderBubble",
        optionsHtml: r.optionsHtml || ""
      });

      // Mark as triggered
      await updateDoc(
        doc(db, "Users", uid, "Reminders", r.id),
        { triggered: true }
      );
    }

  } catch (err) {
    error("checkReminders error:", err);
  }
}
async function handleReminder(text, userContext, forceSave = false) {
  if (!userContext.uid) {
    return "🌙 I can’t store a reminder without knowing who you are.";
  }

  // ------------------------------------
  // 1. PARSE NATURAL LANGUAGE (LOCAL)
  // ------------------------------------
  const parsed = buildParsedReminder(text);

  if (!parsed.trigger) {
    return "🌴 I heard you, but I’m not sure when you want me to remind you.";
  }

  // Load existing context so we can MERGE instead of overwrite
  const ctx = loadVaultContext() || {};

  // ------------------------------------
  // 2. FORCE SAVE MODE (YES button)
  // ------------------------------------
  if (forceSave) {
    try {
      await addDoc(
        collection(db, "Users", userContext.uid, "Reminders"),

          message: parsed.text,
          trigger: parsed.trigger,
          time: parsed.time ? new Date(parsed.time) : null,

          createdAt: new Date()
        }
      );
    } catch (err) {
      error("addReminder error:", err);
      return "🔥 The Vault hesitated… I couldn’t store that reminder just now.";
    }

    // ⭐ MERGE context instead of overwriting
    saveVaultContext({
      ...ctx,
      lastIntent: "reminder",
      lastReminderText: parsed.text,
      unresolved: false,
      pendingQuestion: null,
      pendingData: null
    });

    ]);

  // ------------------------------------
  if (parsed.needsConfirmation) {
    saveVaultContext({
      ...ctx,
      pendingQuestion: "reminder_confirmation",

  // ------------------------------------

      collection(db, "Users", userContext.uid, "Reminders"),
      {
        message: parsed.text,
        trigger: parsed.trigger,
        time: parsed.time ? new Date(parsed.time) : null,
        location: parsed.location || null,
        triggered: false,
        createdAt: new Date()
      }
    );
  } catch (err) {
    error("addReminder error:", err);
    return "🔥 The Vault hesitated… I couldn’t store that reminder just now.";
  }

  // ⭐ MERGE context instead of overwriting
  saveVaultContext({
    ...ctx,
    lastIntent: "reminder",
    lastReminderText: parsed.text,
    unresolved: false,
    pendingQuestion: null,
    pendingData: null
  });

  return pick([

    "🌴 I’ve stored that for the perfect moment.",

  // 🔒 HARD GUARD — prevents ANY duplicate initialization
  if (PulseRealm.__startAppLocked) return;

  // Ensure DOM is fully loaded (mobile fix)
  if (document.readyState !== "complete") {
    window.addEventListener("load", startApp, { once: true });

  if (!identity) {
    await new Promise(r => setTimeout(r, 50));
    return startApp();

  if (appStarted) return;

  // Wait for vault UI to mount
  await new Promise(r => requestAnimationFrame(r));

  initProfileVisuals();

  requestAnimationFrame(() => {
    checkReminders(identity).catch(warn);

  log("StartApp Flags:", {
    vaultStart,
    vaultOpened,
    visibilityArmed,
    appStarted,
    chamberWatcherStarted,
    vaultIntroStarted,
    __vaultInitStarted: PulseRealm.__vaultInitStarted,
    __vaultIntroShown: PulseRealm.__vaultIntroShown
  });
}

        vaultSmoke.classList.remove("smoke-puff");
        vaultSmoke.classList.add("smoke-dissipate");
        document.body.classList.add("vault-post-intro");
      }
    }
  ];

  // Execute timeline
  for (const step of steps) {
    setTimeout(step.fn, step.t);
  }
}

  }
}
function onInputKeydown(e) {
  // ENTER key only
  if (e.key === "Enter") {
    e.preventDefault();   // stop newline
    e.stopPropagation();  // stop bubbling

    // If Shift+Enter, allow newline (optional)
    if (e.shiftKey) {
      return;
    }

    // Trigger send
    onSendClick(e);
  }
}

async function initVaultPage() {
  if (PulseRealm.__vaultInitStarted) return;
  PulseRealm.__vaultInitStarted = true;

  // Attach listeners once
  if (!PulseRealm.__vaultListenersAttached) {
    PulseRealm.__vaultListenersAttached = true;
    sendBtn.onclick = onSendClick;
    inputEl.addEventListener("keydown", onInputKeydown);
    hideBtn.onclick = onHideClick;
  }

  sessionContainer.classList.remove("vaultSessionContainer-reopen");

  vaultStart = nowMs();

  // Preload sounds
  initializeVaultButton();


  // SECURITY FIRST — before heavy loads
  const sec = await fetchSecurityState(identity.bank).catch(() => ({}));

  if (sec.dangerMode || sec.appLocked || sec.requiresPin || sec.vaultLocked) {
    showAppLockScreen();
    if (sec.dangerMode) showDangerOverlay();
    return;
  }

  // Load heavy data in parallel
  const [wallet, loyalty, businesses, events] = await Promise.all([
    loadWallet().catch(() => null),
    loadLoyalty(identity.bank, token).catch(() => null),
    loadBusinesses().catch(() => []),
    loadEvents().catch(() => [])
  ]);

  PulseRealm.userLoyalty = loyalty;

  // Normalize last visit
  PulseRealm.userLoyaltyContext = {
    ...PulseRealm.userLoyaltyContext,
    lastVaultVisit: normalizeDateMs(identity.lastVaultVisit)
  };

  // Build user context
  PulseRealm.userContext = {
    ...identity,
    ...(wallet || {}),
    loyalty,
    saveConversationMemory(update) {
      const ctx = loadVaultContext() || {};
      Object.assign(ctx, update);
      saveVaultContext(ctx);
    }
  };

  // Let layout settle
  await new Promise(r => setTimeout(r, 300));

  // Run intro in parallel
  safeIntro(loyalty);

  // Save last visit (Firestore, not backend)
  queueMicrotask(() => {
    setDoc(
      doc(db, "Users", identity.bank),
      { lastVaultVisit: serverTimestamp() },
      { merge: true }
    ).catch(() => {});
  });

  requestAnimationFrame(() => {
    startApp();
    armVisibility();
  });

  hideTPLoader();
}
async function onSendClick(e) {
  let bubbleDelay = nowMs();
  const text = inputEl.value.trim();
  if (!text) return;

  const minimized = sessionContainer.classList.contains("minimized");
  const userLocation = PulseRealm.userLoyaltyContext.location || null;

  // ───────────────────────────────────────────────
  // Expand or minimize chat
  // ───────────────────────────────────────────────
  if (minimized) {
    closeWallet();

    sessionContainer.classList.remove("minimized");
    sessionSpacer.classList.remove("minimized");
    messageBubbleArea.classList.remove("minimized");

    messageBubbleArea.style.maxHeight = "100%";
    messageBubbleArea.style.height = "auto";

    sessionContainer.classList.add("vaultSessionContainer-reopen");
    sessionSpacer.classList.add("reopen");
    messageBubbleArea.classList.add("reopen");

    // Force layout once
    void sessionContainer.offsetHeight;

    sessionContainer.classList.add("reopenAnim");

    const cleanup = () => {
      sessionContainer.classList.remove("reopenAnim");
      sessionSpacer.classList.remove("reopen");
      sessionContainer.removeEventListener("transitionend", cleanup);
    };

    sessionContainer.addEventListener("transitionend", cleanup);

  } else if (vaultOpened) {
    // Minimize chat
    sessionContainer.classList.add("minimized");
    sessionSpacer.classList.add("minimized");
    messageBubbleArea.classList.add("minimized");

    sessionContainer.classList.remove("vaultSessionContainer-reopen");
    sessionSpacer.classList.remove("reopen");
    messageBubbleArea.classList.remove("reopen");

    openWallet();
  }

  // Clear input
  inputEl.value = "";

  // User bubble
  addBubble("You", text, true);

  // ───────────────────────────────────────────────
  // 1. Numeric choice
  // ───────────────────────────────────────────────
  if (PulseRealm.__pendingPlaceChoices && /^[1-3]$/.test(text)) {
    const response = await handlePlaceChoice(text, userLocation);

    addBubble("Vault Spirit", response.text, false, {
      optionsHtml: response.optionsHtml || ""
    });

    return;
  }

  // ───────────────────────────────────────────────
  // 2. Directions intent
  // ───────────────────────────────────────────────
  const maybeDirections = await handleDirectionsIntent(text, userLocation);

  if (maybeDirections) {
    addBubble("Vault Spirit", maybeDirections.text, false, {
      optionsHtml: maybeDirections.optionsHtml || ""
    });
    return;
  }

  // ───────────────────────────────────────────────
  // 3. Backend AI
  // ───────────────────────────────────────────────
  let reply;
  try {
    reply = await askVaultSpirit(text);
  } catch (err) {
    error("askVaultSpirit failed:", err);
    addBubble("Vault Spirit", "🌺 The winds were noisy… try again.", false);
    return;
  }

  // Render reply next frame
  requestAnimationFrame(() => {
    addBubble("Vault Spirit", reply.text || "", false, {
      optionsHtml: reply.optionsHtml || "",
      extraClasses: reply.extraClasses || ""
    });

    if (reply.pendingQuestion) {
      saveVaultContext({
        pendingQuestion: reply.pendingQuestion,
        pendingData: reply.pendingData || null
      });
    }
  });

  // Timing log
  bubbleDelay = nowMs() - bubbleDelay;
  log("⏱ Bubble appeared after:", bubbleDelay, "ms");

  // Save context (single write)
  saveVaultContext({
    lastIntent: "question",
    lastQuestion: text,
    unresolved: false
  });
}
function onHideClick() {
  const minimized = sessionContainer.classList.contains("minimized");

  if (!minimized) {
    // → MINIMIZE CHAT (batched)
    sessionContainer.classList.add("minimized");
    sessionSpacer.classList.add("minimized");
    messageBubbleArea.classList.add("minimized");

    sessionContainer.classList.remove("vaultSessionContainer-reopen");
    sessionSpacer.classList.remove("reopen");
    messageBubbleArea.classList.remove("reopen");

    // Only open wallet if vault was opened
    if (vaultOpened) openWallet();

    return;
  }

  // → REOPEN CHAT (batched)
  sessionContainer.classList.remove("minimized");
  sessionSpacer.classList.remove("minimized");
  messageBubbleArea.classList.remove("minimized");

  sessionContainer.classList.add("vaultSessionContainer-reopen");
  sessionSpacer.classList.add("reopen");
  messageBubbleArea.classList.add("reopen");

  // Always close wallet when chat reopens
  closeWallet();
}
function tryInitVault(source = "unknown") {
  const now = nowMs();

  // Guard FIRST — no wasted logs, no wasted work
  if (PulseRealm.__vaultInitStarted) {
    log(`⛔ tryInitVault blocked (${source}) — init already started`);
    return;
  }

  log(`🔍 tryInitVault from: ${source}`, {
    now,
    __vaultInitStarted: PulseRealm.__vaultInitStarted
  });

  // No debounce needed — initVaultPage owns the flag
  tryInitVault.lastInitAttempt = now;

  log("🚀 Running initVaultPage (from tryInitVault)");
  initVaultPage();
}
