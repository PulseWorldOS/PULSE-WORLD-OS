async function sendAdminAlertEmail(templateId, payload) {
  try {
    const res = await fetch("/.netlify/functions/PULSE-SERVER-EMAIL", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        templateId,
        payload
      })
    });

    }

  } catch (err) {
    console.error("❌ Failed to call pulse-server-email:", err);
    return { success: false, error: err };
  }

async function loadUsers() {
  try {
    const identityId = PulseRealm.PulseBank;

    // If no bank ID exists, fallback to single identity
    if (!identityId) {
      safeLog("No Pulse Tether — using local identity only");
      usersCache = [];
      await populateUserSelect();
      badgeUsers.textContent = "1";
      return;
    }

    // Try PulseDB
    
    
    if (!PulseRealm.PulseFirebaseDB) {
      safeLog("No PulseDB available — using local identity only");
      usersCache = [];
      await populateUserSelect();
      badgeUsers.textContent = "1";
      return;
    }

    // Query identities belonging to this bank ID
    const identitiesCol = PulseRealm.PulseFirebaseDB.collection("PulseIdentity").get();
    const snapshot = await identitiesCol.find({ identityId });

    // Normalize
    usersCache = Array.isArray(snapshot) ? snapshot : [];

    // If no identities found, fallback to single identity
    if (!usersCache.length) {
      safeLog("No Pulse Tether found — using local identity only");
      safeLog("Please Click Here to Create Tether:" + PulseRealm.PulseBankURL);
      usersCache = [];
      await populateUserSelect();
      badgeUsers.textContent = "1";
      return;
    }

    // Populate dropdown
    await populateUserSelect();
    badgeUsers.textContent = usersCache.length;
    safeLog("Loaded identities for PulseIdentity", { count: usersCache.length });

  } catch (err) {
    safeLog("Failed to load identities", { error: String(err) });
    usersCache = [];
    await populateUserSelect();
    badgeUsers.textContent = "1";
  }
}


async function populateUserSelect() {
  userSelect.innerHTML = "";

  if (!usersCache.length) {
    const opt = document.createElement("option");
    opt.value = PulseRealm.PulseIdentity.email;
    opt.textContent = PulseRealm.PulseIdentity.email;
    userSelect.appendChild(opt);
    return;
  }

  const placeholder = document.createElement("option");
  placeholder.value = "";
  placeholder.textContent = "Select a User Identity…";
  userSelect.appendChild(placeholder);

  usersCache.forEach((u) => {
    const opt = document.createElement("option");
    opt.value = u.id || u.uid || u.email || "";
    opt.textContent = u.email || u.name || u.id || "Unknown User";
    opt.dataset.email = u.email || "";
    opt.dataset.name = u.name || "";
    userSelect.appendChild(opt);
  });
}

function loadTemplates() {
  templatesCache = [];

  Object.keys(PulseRealm.EmailTemplates)
    .sort((a, b) => a.localeCompare(b))   // ⭐ alphabetize keys
    .forEach((key) => {
      const tpl = PulseRealm.EmailTemplates[key];

      if (typeof tpl === "object") {
        templatesCache.push({
          id: key,
          label: key.replace(/([A-Z])/g, " $1").trim(),
          template: tpl
        });
      }
    });

  populateTemplateSelect();
  badgeTemplates.textContent = templatesCache.length;
  safeLog("Loaded templates", { count: templatesCache.length });
}


function populateTemplateSelect() {
  templateSelect.innerHTML = "";

  const placeholder = document.createElement("option");
  placeholder.value = "";
  placeholder.textContent = "Select email template…";
  templateSelect.appendChild(placeholder);

  templatesCache.forEach((t) => {
    const opt = document.createElement("option");
    opt.value = t.id;
    opt.textContent = t.label;
    templateSelect.appendChild(opt);
  });
}

function buildEmailPayload() {  
  const userEmail = PulseRealm.PulseUserEmail ?? "AIOvermindPrime@PulseWorld.Net";
  const Email = PulseRealm.PulseEmail ?? `${stripTimestamp(PulseRealm.PulseUser || "Unknown")}@PulseWorld.Net`;
  
  const toEmail = toInput.value.trim() || null;
  const ccEmail = ccInput.value.trim() || userEmail;
  ccInput.value = userEmail;

  // ---------------------------------------------------------
  // Pull template from templatesCache (alphabetized loader)
  // ---------------------------------------------------------
  const templateEntry = templatesCache.find(t => t.id === selectedTemplate);
  if (!templateEntry) throw new Error("Template not found");

  const tpl = templateEntry.template;

  const userName  = selectedUser ?? PulseRealm.PulseUser ?? "Friend";

  const payload = {
    to: toEmail,
    cc: ccEmail,

    email: Email,
    name: selectedUser,
    user: selectedUser,
    rawPayload: null,
    templateId: selectedTemplate
  };

  // ---------------------------------------------------------
  // AUTO‑GENERATE SUBJECT FROM TEMPLATE (write to UI)
  // ---------------------------------------------------------
  const subject =
    typeof tpl.subject === "function"
      ? tpl.subject(payload)
      : tpl.subject || "";

  subjectInput.value = subject;   // ⭐ WRITE to UI

  // ---------------------------------------------------------
  // AUTO‑GENERATE PAYLOAD JSON (write to UI)
  // ---------------------------------------------------------
  let generatedPayload = "";

  // If template defines an example payload, use it
  if (tpl.examplePayload) {
    generatedPayload = JSON.stringify(tpl.examplePayload, null, 2);
  }

  // If template has no example payload, clear the field
  payloadInput.value = generatedPayload;   // ⭐ WRITE to UI

  // Store raw payload into meta
  payload.rawPayload = generatedPayload;

  // ---------------------------------------------------------
  // HTML
  // ---------------------------------------------------------
  const html =
    typeof tpl.html === "function" ? tpl.html(payload) : tpl.html;

  // ---------------------------------------------------------
  // Text fallback
  // ---------------------------------------------------------
  const text =
    typeof tpl.text === "function" ? tpl.text(payload) : tpl.text || "";

  return {
    email: toEmail,
    cc: ccEmail,
    subject,
    html,
    text,
    meta: payload,
    templateId: selectedTemplate
  };
}

function escapeHtml(str) {
  return String(str || "")
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;");
}
