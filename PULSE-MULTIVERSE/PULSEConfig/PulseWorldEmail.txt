      method: "POST",

async function loadUsers() {
  try {
    const identityId = PulseRealm.PulseBank;

    // If no bank ID exists, fallback to single identity
    if (!identityId) {
      safeLog("No Pulse Tether — using local identity only");
      usersCache = [];
      await populateUserSelect();
      badgeUsers.textContent = "1";
      return;
    }

    // Try PulseDB
    
    
    if (!PulseRealm.PulseFirebaseDB) {
      safeLog("No PulseDB available — using local identity only");
      usersCache = [];
      await populateUserSelect();
      badgeUsers.textContent = "1";
      return;
    }

    // Query identities belonging to this bank ID
    const identitiesCol = PulseRealm.PulseFirebaseDB.collection("PulseIdentity");
    const snapshot = await identitiesCol.find({ identityId });

    // Normalize
    usersCache = Array.isArray(snapshot) ? snapshot : [];

    // If no identities found, fallback to single identity
    if (!usersCache.length) {
      safeLog("No Pulse Tether found — using local identity only");
      safeLog("Please Click Here to Create Tether: <font color=blue><a href=" + PulseRealm.PulseBankURL + ">Pulse Tether</a></font>");
      usersCache = [];
      await populateUserSelect();
      badgeUsers.textContent = "1";
      return;
    }

    // Populate dropdown
    await populateUserSelect();
    badgeUsers.textContent = usersCache.length;
    safeLog("Loaded identities for PulseIdentity", { count: usersCache.length });

  } catch (err) {
    safeLog("Failed to load identities", { error: String(err) });
    usersCache = [];
    await populateUserSelect();
    badgeUsers.textContent = "1";
  }
}

  userSelect.innerHTML = "";
  ccInput.value = PulseRealm.PulseIdentity.userEmail;

    const opt = document.createElement("option");

    opt.value = localEmail;
    opt.textContent = localEmail;

  }

  placeholder.value = PulseRealm.PulseIdentity.email || PulseRealm.PulseIdentity.userEmail || "";
  placeholder.textContent = "Select a User Identity…";
  userSelect.appendChild(placeholder);

    const opt = document.createElement("option");

    opt.dataset.email = u.email || "";
    opt.dataset.name = u.name || "";
    userSelect.appendChild(opt);
  });
}

function loadTemplates() {
  templatesCache = [];

  Object.keys(PulseRealm.EmailTemplates)
    .sort((a, b) => a.localeCompare(b))   // ⭐ alphabetize keys
    .forEach((key) => {
      const tpl = PulseRealm.EmailTemplates[key];

      if (typeof tpl === "object") {
        templatesCache.push({
          id: key,
          label: key.replace(/([A-Z])/g, " $1").trim(),
          template: tpl
        });
      }
    });

  populateTemplateSelect();
  badgeTemplates.textContent = templatesCache.length;
  safeLog("Loaded templates", { count: templatesCache.length });
}

  templatesCache.forEach((t) => {
    const opt = document.createElement("option");

    opt.textContent = t.label;
    templateSelect.appendChild(opt);
  });
}

function getActiveAliasHandle() {
  const chosenVal = selectedUser || userSelect?.value || PulseRealm.PulseIdentity?.name || PulseRealm.PulseUser || "PulseUser";
  const rawHandle = String(chosenVal).includes("@") ? String(chosenVal).split("@")[0] : String(chosenVal);
  return PulseRealm.sanitizeAliasHandle ? PulseRealm.sanitizeAliasHandle(rawHandle) : rawHandle.replace(/\s+/g, "-");
}

function getSelectedDomain() {

}

  if (fromAliasPreview) {

function populateTemplateDefaults() {
  const templateEntry = templatesCache.find(t => t.id === selectedTemplate);
  if (!templateEntry) return;
  const tpl = templateEntry.template;
  const dummy = {
    to: toInput.value.trim(),
    cc: ccInput.value.trim(),
    email: `${getActiveAliasHandle()}@${getSelectedDomain()}`,
    name: PulseRealm.PulseIdentity?.name || PulseRealm.PulseUser || "Friend",
    templateId: selectedTemplate
  };
  const subj = typeof tpl.subject === "function" ? tpl.subject(dummy) : (tpl.subject || "");
  if (subj) subjectInput.value = subj;
  if (tpl.examplePayload) {
    payloadInput.value = JSON.stringify(tpl.examplePayload, null, 2);
  }
}

function buildEmailPayload() {
  const identity = PulseRealm.PulseIdentity || {};
  const fromAlias = getActiveAliasHandle();
  const fromDomain = getSelectedDomain();
  const fromEmail = `${fromAlias}@${fromDomain}`;
  const fromName = identity.name || PulseRealm.PulseUser || fromAlias;

  const toEmail = toInput.value.trim();
  if (!toEmail || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(toEmail)) {
    throw new Error("Please enter a valid recipient email address in the 'To' field.");
  }

  const rawCc = ccInput.value.trim();
  const ccEmail = rawCc && !/^(youremail|unknown)@domain\.com$/i.test(rawCc) ? rawCc : null;

  const userTypedSubject = subjectInput.value.trim();
  const userTypedBody = payloadInput.value.trim();

  let subject = userTypedSubject || `Message from ${fromName} (${fromEmail})`;
  let html = "";
  let text = userTypedBody;
  let templateId = selectedTemplate || "PulseCustomMessage";

  if (selectedTemplate) {
    const templateEntry = templatesCache.find(t => t.id === selectedTemplate);
    if (templateEntry) {
      const tpl = templateEntry.template;
      let parsedRaw = userTypedBody;
      try { if (userTypedBody.startsWith("{")) parsedRaw = JSON.parse(userTypedBody); } catch {}
      const tplData = {
        to: toEmail,
        cc: ccEmail,
        email: fromEmail,
        name: fromName,
        user: fromName,
        rawPayload: parsedRaw,
        templateId: selectedTemplate
      };
      subject = userTypedSubject || (typeof tpl.subject === "function" ? tpl.subject(tplData) : tpl.subject || subject);
      html = typeof tpl.html === "function" ? tpl.html(tplData) : (tpl.html || "");
      text = typeof tpl.text === "function" ? tpl.text(tplData) : (tpl.text || userTypedBody);
    }
  }

  if (!html) {
    if (!userTypedBody) {
      throw new Error("Please write a message or select an email template.");
    }
    html = `<p style="white-space:pre-wrap;">${escapeHtml(userTypedBody)}</p>`;
  }

  return {
    mode: selectedTemplate ? "admin" : "user",
    to: toEmail,
    email: toEmail,
    cc: ccEmail,
    fromAlias,
    fromDomain,
    fromName,
    localId: identity.id || null,
    stripeID: identity.tokenID || identity.stripeID || PulseRealm.PulseBankID || null,
    subject,
    html,
    text: text || userTypedBody || subject,
    templateId
  };
}

        <h2 style="margin-top:0;">Preview: ${escapeHtml(subjectInput.value.trim())}</h2>
        <p><strong>To:</strong> ${escapeHtml(toInput.value.trim())}</p>
        <p><strong>CC:</strong> ${escapeHtml(ccInput.value.trim())}</p>
        <p><strong>Template:</strong> ${escapeHtml(payload.templateId)}</p>
        <hr style="border:none;border-top:1px solid rgba(255,255,255,0.15);margin:12px 0;">
        <div>${payload.html}</div>
      </div>
    `;

    showModal(previewHtml);
    safeLog("Preview generated", { to: payload.email, template: payload.templateId });

  } catch (err) {
    
  }
}

function escapeHtml(str) {
  return String(str || "")
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;");
}

async function handleSend() {
  try {
    setLoading(true);
    const payload = buildEmailPayload();

    const result = await sendAdminAlertEmail(payload.templateId, payload);

    safeLog("Email sent", { to: payload.email, template: payload.templateId, result });

  } catch (err) {
    safeLog("Failed to send email", { error: String(err) });
  } finally {
    setLoading(false);
  }
}
