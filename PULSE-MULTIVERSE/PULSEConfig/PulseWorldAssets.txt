/* ============================================================
   ETHERS LOADER
   ============================================================ */

async function loadEthers() {

    try {

        const module = await import(
            "https://cdn.jsdelivr.net/npm/ethers@6.13.4/dist/ethers.min.js"
        );

        window.ethers = module;

        PulseWorldState.ethersLoaded = true;

        log(
            "%c✅ ethers loaded",
            "color:#00ffaa"
        );

        return true;

    }
    catch(error){

        console.warn(
            "[PulseWorld] ethers unavailable",
            error
        );

        return false;
    }
}

/* ============================================================
   ASSET DISCOVERY
   ============================================================ */

function scanAssets(){

    const summary = {

        categories : categoryBlocks.length,

        cards      : assetCards.length,

        galleries  : galleryImages.length,

        showcases  : showcaseImages.length

    };

    console.table(summary);

    return summary;
}

/* ============================================================
   UI EFFECTS
   ============================================================ */

function installCardEffects(){

    assetCards.forEach(card=>{

        card.addEventListener("mouseenter",()=>{

            card.style.transform =
                "translateY(-6px)";

            card.style.transition =
                "all .25s ease";

            card.style.boxShadow =
                "0 0 40px rgba(0,234,255,.35)";
        });

        card.addEventListener("mouseleave",()=>{

            card.style.transform =
                "translateY(0px)";

            card.style.boxShadow =
                "";
        });

    });

}

function installImageEffects(){

    galleryImages.forEach(image=>{

        image.loading = "lazy";

        image.addEventListener("click",()=>{

            window.open(
                image.src,
                "_blank"
            );

        });
    });

}

/* ============================================================
   CATEGORY METADATA
   ============================================================ */

function attachCategoryData(){

    categoryBlocks.forEach((block,index)=>{

        block.dataset.categoryIndex =
            index + 1;

        const title =
            block.querySelector("h2");

        if(title){

            block.dataset.category =
                title.textContent.trim();
        }

    });

}

/* ============================================================
   METAMASK
   ============================================================ */

async function connectMetaMask(){

    if(!window.ethereum){

        console.warn(
            "[PulseWorld] MetaMask not detected"
        );

        return null;
    }

    try{

        const accounts =
            await window.ethereum.request({
                method:"eth_requestAccounts"
            });

        const address =
            accounts?.[0] ?? null;

        const chainId =
            await window.ethereum.request({
                method:"eth_chainId"
            });

        PulseWorldState.connectedWallet =
            address;

        PulseWorldState.chainId =
            chainId;

        log(
            `%c🔗 MetaMask Connected ${address}`,
            "color:#38bdf8"
        );

        return {
            address,
            chainId
        };

    }
    catch(error){

        console.error(
            "[PulseWorld] MetaMask connect failed",
            error
        );

        return null;
    }
}

/* ============================================================
   PULSE ASSET SYSTEM
   ============================================================ */

async function initializePulseAssets(){

    try{

        if(
            globalThis.PulseRealm &&
            globalThis.PulseRealm.PulseAssets
        ){

            const assets =
                PulseRealm.PulseAssets.getAssets?.() || {};

            PulseWorldState.pulseAssets =
                assets;

            log(
                "%c✅ Pulse Assets Attached",
                "color:#00ffaa"
            );

            return assets;
        }

        log(
            "%c⚠ PulseAssets unavailable",
            "color:gold"
        );

        return null;

    }
    catch(error){

        console.error(
            "[PulseWorld] Asset initialization failed",
            error
        );

        return null;
    }

}

/* ============================================================
   NETWORK WATCHERS
   ============================================================ */

function installEthereumListeners(){

    if(!window.ethereum)
        return;

    ethereum.on(
        "accountsChanged",
        accounts=>{

            PulseWorldState.connectedWallet =
                accounts?.[0] || null;

            log(
                "👤 Wallet Changed",
                PulseWorldState.connectedWallet
            );
        }
    );

    ethereum.on(
        "chainChanged",
        chainId=>{

            PulseWorldState.chainId =
                chainId;

            log(
                "🌐 Chain Changed",
                chainId
            );
        }
    );

}
