  function setupPhotoChange(btn, input, imgEl, realmKey) {

    // Use PulseRealm as truth, NOT the DOM (DOM may contain corrupted blob)
    const oldPhoto = PulseRealm[realmKey] || imgEl.src;

    btn.onclick = () => input.click();

    input.addEventListener("change", async (event) => {
      const file = event.target.files[0];

      reader.onload = async (e) => {
        let result = e.target.result;

        // ⭐ Normalize data URL (Chrome adds garbage on refresh)
        result = result
          .replace(/\s+/g, "")          // remove whitespace
          .trim();

        // ⭐ Accept ANY valid image data URL (png, jpg, webp, gif, svg)

        // ⭐ Save stable version
        PulseRealm[realmKey] = result;

        // ⭐ Apply to DOM
        imgEl.src = result;

        PulseRealm.PulseWorldPopup("Photo Updated", "Undo", () => {

          imgEl.src = oldPhoto;
        });
      };

    });
