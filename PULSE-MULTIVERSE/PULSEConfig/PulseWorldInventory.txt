  function setupPhotoChange(btn, input, imgEl, realmKey) {
    btn.onclick = () => input.click();

    input.addEventListener("change", async (event) => {
      const file = event.target.files[0];
      if (!file) return;

        let result = e.target.result;

        // ⭐ Normalize data URL (Chrome adds garbage on refresh)
        result = result
          .replace(/\s+/g, "")          // remove whitespace
          .trim();

        // ⭐ Accept ANY valid image data URL (png, jpg, webp, gif, svg)
        if (!/^data:image\//i.test(result)) {
          PulseRealm.PulseWorldPopup("Invalid image format", "OK", () => {});
          return;
        }

        // ⭐ Save stable version
        PulseRealm[realmKey] = result;
        imgEl.src = result;

        const identityPhotoKey = {
          PulsePhoto: "photoURL",
          PulseAliasPhoto: "aliasPhotoURL",
          PulseBizPhoto: "bizphotoURL",
          PulseBizAliasPhoto: "bizaliasPhotoURL"

async function savebutton() {
  console.log("[SaveButton] Starting savebutton()…");

  const saveBtnEl = document.getElementById("saveBtn");
  const msgEl = document.getElementById("msg");
  const displayNameStatus = document.getElementById("displayNameStatus");

  const previousIdentity = PulseRealm.PulseIdentityStore.readJSON("PulseIdentity");
  const previousPhotos = PulseRealm.PulseIdentityStore.readJSON("PulsePhotos");
  const fullNameValue = document.getElementById("fullName")?.value.trim();
  const displayNameValue = document.getElementById("displayNameInput")?.value.trim();
  const aliasDomainValue = document.getElementById("aliasDomainSelect")?.value || previousIdentity.emailDomain || "pulseworld.net";
  const emailValue = document.getElementById("email")?.value.trim().toLowerCase();
  const phoneValue = document.getElementById("phone")?.value.trim();
  const countryValue = document.getElementById("country")?.value || "US";
  const roleValue = document.getElementById("role")?.value || previousIdentity.role || "User";
  const smsOptValue = document.getElementById("smsOpt")?.checked ?? previousIdentity.smsOpt ?? false;
  const emailOptValue = document.getElementById("emailOpt")?.checked ?? previousIdentity.emailOpt ?? true;
  const trustedDevice = document.getElementById("stayToggle")?.checked;

  const targetName = displayNameValue || previousIdentity.name;

  if (saveBtnEl) {
    saveBtnEl.disabled = true;
    saveBtnEl.textContent = "Verifying & Saving…";
  }

  try {
    // 1. Server-side uniqueness check before saving
    if (targetName && typeof PulseRealm.checkIdentityName === "function") {
      const check = await PulseRealm.checkIdentityName(targetName, {
        emailDomain: aliasDomainValue,
        localId: previousIdentity.id,
        stripeID: previousIdentity.tokenID || previousIdentity.stripeID || null
      });

      if (check.ok && check.available === false) {
        const sugg = Array.isArray(check.suggestions) && check.suggestions.length
          ? ` Try: ${check.suggestions.join(", ")}`
          : "";
        if (displayNameStatus) {
          displayNameStatus.style.display = "block";
          displayNameStatus.style.color = "#ff6b6b";
          displayNameStatus.textContent = `✖ ${check.error || "Display Name already taken."}${sugg}`;
        }
        if (msgEl) {
          msgEl.style.color = "#ff6b6b";
          msgEl.textContent = check.error || "Display Name is already taken.";
        }
        PulseRealm.PulseWorldPopup("Display Name Already Taken — Please Choose Another", "OK", () => {});
        return;
      }
    }

    // 2. Persist locally (updating alias email with chosen PulseWorld domain)
    const computedAliasEmail = PulseRealm.buildAliasEmail
      ? PulseRealm.buildAliasEmail(targetName, aliasDomainValue)
      : `${targetName}@${aliasDomainValue}`;

    PulseRealm.PulseUserName = fullNameValue || PulseRealm.PulseUserName;
    PulseRealm.PulseUser = targetName || PulseRealm.PulseUser;
    PulseRealm.PulseEmailDomain = aliasDomainValue;
    PulseRealm.PulseEmail = computedAliasEmail;
    PulseRealm.PulseUserEmail = emailValue || PulseRealm.PulseUserEmail;
    PulseRealm.PulsePhone = phoneValue || PulseRealm.PulsePhone;
    PulseRealm.PulseCountry = countryValue || PulseRealm.PulseCountry;
    PulseRealm.PulseRole = roleValue;
    PulseRealm.PulseTrustedDevice = trustedDevice ?? PulseRealm.PulseTrustedDevice;

    const savedIdentity = PulseRealm.PulseIdentityStore.save({
      name: targetName,
      userName: fullNameValue || previousIdentity.userName,
      emailDomain: aliasDomainValue,
      email: computedAliasEmail,
      aliasEmail: computedAliasEmail,
      userEmail: emailValue || previousIdentity.userEmail,
      phone: phoneValue || previousIdentity.phone,
      country: countryValue || previousIdentity.country,
      role: roleValue,
      smsOpt: smsOptValue,
      emailOpt: emailOptValue,
      device: trustedDevice ?? previousIdentity.device
    });

    // 3. Immediately sync to Supabase (with server-side unique enforcement)
    let syncRes = null;
    if (typeof PulseRealm.pulseSupabaseSync === "function") {
      syncRes = await PulseRealm.pulseSupabaseSync({ force: true, enforceUnique: true, skipGeo: true });
      if (syncRes && syncRes.duplicate) {
        const restored = PulseRealm.PulseIdentityStore.restore(previousIdentity, previousPhotos);
        renderDashboard(restored);
        PulseRealm.PulseWorldPopup("Name Already Claimed — Restored Previous Identity", "OK", () => {});
        return;
      }
    }

    // 4. Auto-create/link Stripe Connect Express if user entered a valid real email & has no Stripe ID yet
    const hasValidRealEmail = emailValue && /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(emailValue) && !/^(youremail|unknown)@domain\.com$/i.test(emailValue);
    const hasStripeId = typeof savedIdentity.tokenID === "string" && /^acct_[A-Za-z0-9]+$/.test(savedIdentity.tokenID);
    let stripeOnboardingUrl = null;
    let linkedStripeId = savedIdentity.tokenID || null;

    if (hasValidRealEmail && !hasStripeId) {
      try {
        const origin = location.origin;
        const stripeRes = await fetch("/.netlify/functions/PULSE-SERVER-BANK-ACCOUNT", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            action: "createOrLink",
            email: emailValue,
            name: savedIdentity.name,
            localId: savedIdentity.id,
            emailDomain: aliasDomainValue,
            phone: phoneValue,
            country: countryValue,
            role: roleValue,
            host: location.hostname,
            origin,
            returnUrl: `${origin}/?Impulse=PulseWorldInventory&stripe=return`,
            refreshUrl: `${origin}/?Impulse=PulseWorldInventory&stripe=refresh`
          })
        });
        const stripeData = await stripeRes.json();
        if (stripeRes.ok && stripeData.ok && stripeData.stripeID) {
          linkedStripeId = stripeData.stripeID;
          stripeOnboardingUrl = stripeData.onboardingUrl || null;
          PulseRealm.PulseBankID = linkedStripeId;
          PulseRealm.PulseBank = linkedStripeId;
          PulseRealm.PulseIdentityStore.save({
            tokenID: linkedStripeId,
            stripeID: linkedStripeId,
            bank: linkedStripeId,
            stripeStatus: {
              stripeID: linkedStripeId,
              detailsSubmitted: Boolean(stripeData.detailsSubmitted),
              chargesEnabled: Boolean(stripeData.chargesEnabled),
              payoutsEnabled: Boolean(stripeData.payoutsEnabled)
            }
          });
        }
      } catch (stripeErr) {
        console.warn("[SaveButton] Stripe Connect link skipped:", stripeErr.message);
      }
    }

    renderDashboard(PulseRealm.PulseIdentityStore.readJSON("PulseIdentity"));
    console.log("[SaveButton] Identity + Supabase + Stripe sync complete.");

    if (stripeOnboardingUrl) {
      PulseRealm.PulseWorldPopup(
        `Dashboard Saved & Stripe ID Created (${linkedStripeId})! Complete Stripe Onboarding now?`,
        "Open Stripe Onboarding",
        () => window.open(stripeOnboardingUrl, "_blank", "noopener")
      );
    } else {
      PulseRealm.PulseWorldPopup("Dashboard Saved", "Undo", () => {
        const restoredIdentity = PulseRealm.PulseIdentityStore.restore(previousIdentity, previousPhotos);
        renderDashboard(restoredIdentity);
      });
    }
  } finally {
    if (saveBtnEl) {
      saveBtnEl.disabled = false;
      saveBtnEl.textContent = "Save Dashboard";
    }
  }
}
