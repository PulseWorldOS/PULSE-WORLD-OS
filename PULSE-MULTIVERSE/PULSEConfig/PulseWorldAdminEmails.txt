function updateAdminFromPreview() {

async function loadUsers() {
  try {
    // Load real synced users directly from Supabase via PULSE-SERVER-SQL
    const res = await fetch("/.netlify/functions/PULSE-SERVER-SQL", {
      method: "POST",
      headers: { "Content-Type": "application/json", "x-pulse-mode": "admin" },
      body: JSON.stringify({ action: "read" })
    });
    const data = await res.json();
    if (res.ok && data.ok && Array.isArray(data.users) && data.users.length) {
      usersCache = data.users;
      await populateUserSelect();
      badgeUsers.textContent = usersCache.length;
      safeLog("Loaded PulseIdentity users from Supabase", { count: usersCache.length });
      return;
    }

    safeLog("No users in Supabase yet — falling back to local identity");
    usersCache = [];
    await populateUserSelect();
    badgeUsers.textContent = "1";
  } catch (err) {
    safeLog("Failed to load Supabase identities", { error: String(err) });
    usersCache = [];
    await populateUserSelect();
    badgeUsers.textContent = "1";
  }
}


async function populateUserSelect() {
  userSelect.innerHTML = "";

  if (!usersCache.length) {
    const opt = document.createElement("option");
    const localEmail = PulseRealm.PulseIdentity?.userEmail || PulseRealm.PulseIdentity?.email || "";
    opt.value = localEmail;
    opt.textContent = `${PulseRealm.PulseIdentity?.name || "Local User"} (${localEmail})`;
    userSelect.appendChild(opt);
    return;
  }

  const placeholder = document.createElement("option");
  placeholder.value = "";
  placeholder.textContent = "Select a PulseWorld User…";
  userSelect.appendChild(placeholder);

  usersCache.forEach((u) => {
    const opt = document.createElement("option");
    const targetEmail = (u.userEmail && !/^(youremail|unknown)@domain\.com$/i.test(u.userEmail))
      ? u.userEmail
      : (u.aliasEmail || u.email || "");
    opt.value = u.userID || u.id || targetEmail;
    opt.textContent = `${u.name || "Unknown"} — ${u.aliasEmail || u.email || "no-alias"}${u.stripeID ? " [" + u.stripeID + "]" : ""}`;
    opt.dataset.email = targetEmail;
    opt.dataset.name = u.name || "";
    userSelect.appendChild(opt);
  });
}

function loadTemplates() {
  templatesCache = [];

  Object.keys(PulseRealm.EmailTemplates)
    .sort((a, b) => a.localeCompare(b))   // ⭐ alphabetize keys
    .forEach((key) => {
      const tpl = PulseRealm.EmailTemplates[key];

      if (typeof tpl === "object") {
        templatesCache.push({
          id: key,
          label: key.replace(/([A-Z])/g, " $1").trim(),
          template: tpl
        });
      }
    });

  populateTemplateSelect();
  badgeTemplates.textContent = templatesCache.length;
  safeLog("Loaded templates", { count: templatesCache.length });
}

function populateTemplateSelect() {
  templateSelect.innerHTML = "";

  const placeholder = document.createElement("option");
  placeholder.value = "";
  placeholder.textContent = "Select Email Template…";
  templateSelect.appendChild(placeholder);

  templatesCache.forEach((t) => {
    const opt = document.createElement("option");
    opt.value = t.id;
    opt.textContent = t.label;
    templateSelect.appendChild(opt);
  });
}

function buildEmailPayload() {
  const fromAlias = PulseRealm.sanitizeAliasHandle
    ? PulseRealm.sanitizeAliasHandle(adminAliasInput?.value || "AIOvermindPrime")
    : (adminAliasInput?.value || "AIOvermindPrime").trim();
  const fromDomain = fromDomainSelect?.value || "pulseworld.net";

  const recipientEmail =
    toInput?.value?.trim() ||
    selectedUser?.userEmail ||
    selectedUser?.aliasEmail ||
    selectedUser?.email ||
    "";

  if (!recipientEmail || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(recipientEmail)) {
    throw new Error("Select a user or enter a valid recipient email in Direct Recipient Email.");
  }

  const userName = selectedUser?.name || recipientEmail;
  const rawText = payloadInput.value.trim();
  let parsedRaw = rawText || null;
  if (rawText && rawText.startsWith("{")) {
    try { parsedRaw = JSON.parse(rawText); } catch (err) {
      safeLog("Payload treated as plain text (not JSON)");
    }
  }

  let subject = subjectInput.value.trim() || `PulseWorld Admin Notice`;
  let html = "";
  let text = rawText;

  if (selectedTemplate) {
    const templateEntry = templatesCache.find((t) => t.id === selectedTemplate);
    if (templateEntry) {
      const tpl = templateEntry.template;
      const tplPayload = {
        email: recipientEmail,
        name: userName,
        user: selectedUser,
        rawPayload: parsedRaw,
        templateId: selectedTemplate
      };
      subject = subjectInput.value.trim() || (typeof tpl.subject === "function" ? tpl.subject(tplPayload) : tpl.subject || subject);
      html = typeof tpl.html === "function" ? tpl.html(tplPayload) : (tpl.html || "");
      text = typeof tpl.text === "function" ? tpl.text(tplPayload) : (tpl.text || rawText);
    }
  }

  if (!html) {
    if (!rawText) throw new Error("Please enter message content or select a template.");
    html = `<p style="white-space:pre-wrap;">${escapeHtml(rawText)}</p>`;
  }

  return {
    mode: "admin",
    to: recipientEmail,
    email: recipientEmail,
    cc: ccInput?.value?.trim() || null,
    fromAlias,
    fromDomain,
    fromName: `PulseWorld (${fromAlias})`,
    subject,
    html,
    text: text || rawText || subject,
    templateId: selectedTemplate || "PulseAdminNotice"
  };
}

        <div>${payload.html}</div>
      </div>
    `;

    showModal(previewHtml);
    safeLog("Preview generated", { to: payload.email, template: payload.templateId });

  } catch (err) {
    safeLog("Failed to build preview", { error: String(err) });
  }
}

function escapeHtml(str) {
  return String(str || "")
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;");
}

async function handleSend() {
  try {
    setLoading(true);
    const payload = buildEmailPayload();

    const res = await fetch("/.netlify/functions/PULSE-SERVER-EMAIL", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        mode: "admin",
        templateId: payload.templateId,
        fromAlias: payload.fromAlias,
        fromDomain: payload.fromDomain,
        fromName: payload.fromName,
        payload
      })
    });
    const result = await res.json();

    if (res.ok && (result.ok || result.success)) {
      safeLog("Admin email sent", { from: result.from, to: result.to, messageId: result.messageId });
    } else {
      safeLog("Admin email failed", { error: result.error || `HTTP ${res.status}` });
    }

  } catch (err) {
    safeLog("Failed to send email", { error: err.message || String(err) });
  } finally {
    setLoading(false);
  }
}
